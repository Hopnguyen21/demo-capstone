-- ============================================================
-- SMARTFARM DATABASE
-- PostgreSQL + PostGIS
-- ============================================================

-- ============================================================
-- 0. EXTENSIONS
-- ============================================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS postgis;


-- ============================================================
-- 1. USERS / RBAC / TENANT
-- ============================================================

CREATE TABLE tenants (
    tenant_id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(150) NOT NULL,
    code                VARCHAR(50) NOT NULL UNIQUE,
    email               VARCHAR(150),
    phone               VARCHAR(30),
    address             TEXT,
    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE app_users (
    user_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id           UUID REFERENCES tenants(tenant_id) ON DELETE SET NULL,
    username            VARCHAR(100) NOT NULL UNIQUE,
    email               VARCHAR(150) NOT NULL UNIQUE,
    password_hash       TEXT NOT NULL,
    full_name           VARCHAR(150) NOT NULL,
    phone               VARCHAR(30),
    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',
    last_login_at       TIMESTAMPTZ,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE roles (
    role_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(50) NOT NULL UNIQUE,
    description         TEXT
);

CREATE TABLE permissions (
    permission_id       UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(100) NOT NULL UNIQUE,
    description         TEXT
);

CREATE TABLE user_roles (
    user_id             UUID NOT NULL REFERENCES app_users(user_id) ON DELETE CASCADE,
    role_id             UUID NOT NULL REFERENCES roles(role_id) ON DELETE CASCADE,
    assigned_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    PRIMARY KEY (user_id, role_id)
);

CREATE TABLE role_permissions (
    role_id             UUID NOT NULL REFERENCES roles(role_id) ON DELETE CASCADE,
    permission_id       UUID NOT NULL REFERENCES permissions(permission_id) ON DELETE CASCADE,

    PRIMARY KEY (role_id, permission_id)
);


-- ============================================================
-- 2. FARM / FIELD / ZONE
-- ============================================================

CREATE TABLE farms (
    farm_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id           UUID NOT NULL REFERENCES tenants(tenant_id) ON DELETE CASCADE,
    name                VARCHAR(150) NOT NULL,
    description         TEXT,
    address             TEXT,

    -- Farm boundary on map
    boundary            geometry(Polygon, 4326),

    area_m2             NUMERIC(14,2),

    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE fields (
    field_id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,
    name                VARCHAR(150) NOT NULL,
    description         TEXT,

    -- Field boundary
    boundary            geometry(Polygon, 4326),

    area_m2             NUMERIC(14,2),

    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE zones (
    zone_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    field_id            UUID NOT NULL REFERENCES fields(field_id) ON DELETE CASCADE,
    name                VARCHAR(150) NOT NULL,
    description         TEXT,

    -- Zone boundary
    boundary            geometry(Polygon, 4326),

    area_m2             NUMERIC(14,2),

    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 3. EMPLOYEE / FARM WORKER
-- ============================================================

CREATE TABLE employees (
    employee_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id           UUID NOT NULL REFERENCES tenants(tenant_id) ON DELETE CASCADE,
    user_id             UUID UNIQUE REFERENCES app_users(user_id) ON DELETE SET NULL,

    employee_code       VARCHAR(50) NOT NULL,
    full_name           VARCHAR(150) NOT NULL,
    phone               VARCHAR(30),
    email               VARCHAR(150),
    position            VARCHAR(100),

    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',

    joined_at           DATE,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE (tenant_id, employee_code)
);

CREATE TABLE employee_farm_access (
    employee_id         UUID NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    assigned_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    PRIMARY KEY (employee_id, farm_id)
);

CREATE TABLE employee_zone_access (
    employee_id         UUID NOT NULL REFERENCES employees(employee_id) ON DELETE CASCADE,
    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,

    assigned_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    PRIMARY KEY (employee_id, zone_id)
);


-- ============================================================
-- 4. CROP / VARIETY / GROWTH PROFILE
-- ============================================================

CREATE TABLE crops (
    crop_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name                VARCHAR(150) NOT NULL UNIQUE,
    scientific_name     VARCHAR(150),
    description         TEXT,
    is_system_defined   BOOLEAN NOT NULL DEFAULT TRUE,
    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE crop_varieties (
    variety_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    crop_id             UUID NOT NULL REFERENCES crops(crop_id) ON DELETE CASCADE,
    name                VARCHAR(150) NOT NULL,
    description         TEXT,
    is_system_defined   BOOLEAN NOT NULL DEFAULT TRUE,
    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE (crop_id, name)
);

CREATE TABLE growth_profiles (
    growth_profile_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    crop_id             UUID NOT NULL REFERENCES crops(crop_id) ON DELETE CASCADE,
    variety_id          UUID REFERENCES crop_varieties(variety_id) ON DELETE CASCADE,

    name                VARCHAR(150) NOT NULL,
    description         TEXT,

    is_default          BOOLEAN NOT NULL DEFAULT FALSE,
    is_system_defined   BOOLEAN NOT NULL DEFAULT TRUE,

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE growth_stages (
    growth_stage_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    growth_profile_id   UUID NOT NULL REFERENCES growth_profiles(growth_profile_id) ON DELETE CASCADE,

    name                VARCHAR(150) NOT NULL,
    stage_order         INT NOT NULL,
    description         TEXT,

    duration_days       INT,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE (growth_profile_id, stage_order)
);

CREATE TABLE environmental_requirements (
    requirement_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    growth_stage_id     UUID NOT NULL REFERENCES growth_stages(growth_stage_id) ON DELETE CASCADE,

    parameter_code      VARCHAR(50) NOT NULL,

    min_value           NUMERIC(12,4),
    max_value           NUMERIC(12,4),
    target_value        NUMERIC(12,4),

    unit                VARCHAR(30),

    description         TEXT,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Example parameter_code:
-- SOIL_MOISTURE
-- TEMPERATURE
-- AIR_HUMIDITY
-- LIGHT_INTENSITY
-- PH


-- ============================================================
-- 5. PLANTING SEASON / CROP IN ZONE
-- ============================================================

CREATE TABLE planting_seasons (
    planting_season_id  UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,
    crop_id             UUID NOT NULL REFERENCES crops(crop_id) ON DELETE RESTRICT,
    variety_id          UUID REFERENCES crop_varieties(variety_id) ON DELETE RESTRICT,
    growth_profile_id   UUID REFERENCES growth_profiles(growth_profile_id) ON DELETE RESTRICT,

    name                VARCHAR(150) NOT NULL,

    start_date          DATE NOT NULL,
    expected_end_date   DATE,
    actual_end_date     DATE,

    current_growth_stage_id
                        UUID REFERENCES growth_stages(growth_stage_id)
                        ON DELETE SET NULL,

    status              VARCHAR(30) NOT NULL DEFAULT 'PLANNED',

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 6. DEVICE CATALOG
-- ============================================================

CREATE TABLE device_types (
    device_type_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name                VARCHAR(100) NOT NULL UNIQUE,
    category            VARCHAR(30) NOT NULL,
    description         TEXT,

    manufacturer        VARCHAR(100),
    model               VARCHAR(100),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- category:
-- GATEWAY
-- SENSOR
-- ACTUATOR
-- NODE


CREATE TABLE measurement_types (
    measurement_type_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    code                VARCHAR(50) NOT NULL UNIQUE,
    name                VARCHAR(100) NOT NULL,
    unit                VARCHAR(30),
    description         TEXT
);


CREATE TABLE device_type_measurements (
    device_type_id      UUID NOT NULL REFERENCES device_types(device_type_id) ON DELETE CASCADE,
    measurement_type_id UUID NOT NULL REFERENCES measurement_types(measurement_type_id) ON DELETE CASCADE,

    PRIMARY KEY (device_type_id, measurement_type_id)
);


-- ============================================================
-- 7. GATEWAY / NODE / SENSOR / ACTUATOR
-- ============================================================

CREATE TABLE gateways (
    gateway_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    device_type_id      UUID REFERENCES device_types(device_type_id) ON DELETE SET NULL,

    name                VARCHAR(150) NOT NULL,
    device_code         VARCHAR(100) NOT NULL UNIQUE,

    serial_number       VARCHAR(100),
    mac_address         VARCHAR(100),

    latitude            NUMERIC(10,7),
    longitude           NUMERIC(10,7),

    status              VARCHAR(30) NOT NULL DEFAULT 'OFFLINE',

    firmware_version    VARCHAR(50),

    last_seen_at        TIMESTAMPTZ,

    installed_at        TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE sensor_nodes (
    node_id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    gateway_id          UUID REFERENCES gateways(gateway_id) ON DELETE SET NULL,
    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,
    field_id            UUID REFERENCES fields(field_id) ON DELETE SET NULL,
    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,

    device_type_id      UUID REFERENCES device_types(device_type_id) ON DELETE SET NULL,

    name                VARCHAR(150) NOT NULL,
    node_code           VARCHAR(100) NOT NULL UNIQUE,

    serial_number       VARCHAR(100),

    latitude            NUMERIC(10,7),
    longitude           NUMERIC(10,7),

    status              VARCHAR(30) NOT NULL DEFAULT 'OFFLINE',

    firmware_version    VARCHAR(50),

    battery_level       NUMERIC(5,2),

    last_seen_at        TIMESTAMPTZ,
    installed_at        TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE sensors (
    sensor_id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    node_id             UUID NOT NULL REFERENCES sensor_nodes(node_id) ON DELETE CASCADE,

    device_type_id      UUID REFERENCES device_types(device_type_id) ON DELETE SET NULL,

    measurement_type_id UUID NOT NULL REFERENCES measurement_types(measurement_type_id) ON DELETE RESTRICT,

    name                VARCHAR(150) NOT NULL,
    sensor_code         VARCHAR(100) NOT NULL UNIQUE,

    unit                VARCHAR(30),

    status              VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',

    min_reading         NUMERIC(14,4),
    max_reading         NUMERIC(14,4),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE actuators (
    actuator_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    node_id             UUID REFERENCES sensor_nodes(node_id) ON DELETE SET NULL,
    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,
    field_id            UUID REFERENCES fields(field_id) ON DELETE SET NULL,
    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,

    device_type_id      UUID REFERENCES device_types(device_type_id) ON DELETE SET NULL,

    name                VARCHAR(150) NOT NULL,
    actuator_code       VARCHAR(100) NOT NULL UNIQUE,

    actuator_type       VARCHAR(50) NOT NULL,

    status              VARCHAR(30) NOT NULL DEFAULT 'OFFLINE',

    last_seen_at        TIMESTAMPTZ,

    installed_at        TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- actuator_type examples:
-- PUMP
-- VALVE
-- FAN
-- GROW_LIGHT


-- ============================================================
-- 8. DEVICE DEPLOYMENT / INSTALLATION
-- ============================================================

CREATE TABLE deployment_requests (
    deployment_request_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    requested_by        UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    request_date        TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    status              VARCHAR(30) NOT NULL DEFAULT 'SUBMITTED',

    notes               TEXT,

    technician_id       UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    inspected_at        TIMESTAMPTZ,
    confirmed_at        TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE deployment_request_items (
    deployment_request_item_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    deployment_request_id UUID NOT NULL
        REFERENCES deployment_requests(deployment_request_id)
        ON DELETE CASCADE,

    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,

    device_type_id      UUID NOT NULL REFERENCES device_types(device_type_id) ON DELETE RESTRICT,

    required_quantity   INT NOT NULL DEFAULT 1,
    proposed_quantity   INT,
    confirmed_quantity  INT,

    status              VARCHAR(30) NOT NULL DEFAULT 'PROPOSED',

    notes               TEXT
);


-- ============================================================
-- 9. SENSOR READINGS / ENVIRONMENT MONITORING
-- ============================================================

CREATE TABLE sensor_readings (
    reading_id          BIGSERIAL PRIMARY KEY,

    tenant_id           UUID REFERENCES tenants(tenant_id) ON DELETE SET NULL,
    farm_id             UUID REFERENCES farms(farm_id) ON DELETE SET NULL,
    field_id            UUID REFERENCES fields(field_id) ON DELETE SET NULL,
    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,

    node_id             UUID REFERENCES sensor_nodes(node_id) ON DELETE SET NULL,
    sensor_id           UUID NOT NULL REFERENCES sensors(sensor_id) ON DELETE CASCADE,

    measurement_type_id UUID REFERENCES measurement_types(measurement_type_id) ON DELETE SET NULL,

    recorded_at         TIMESTAMPTZ NOT NULL,

    value               NUMERIC(14,4) NOT NULL,

    unit                VARCHAR(30),

    quality_status      VARCHAR(30) DEFAULT 'VALID',

    received_at         TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_sensor_readings_sensor_time
    ON sensor_readings(sensor_id, recorded_at DESC);

CREATE INDEX idx_sensor_readings_zone_time
    ON sensor_readings(zone_id, recorded_at DESC);


-- ============================================================
-- 10. ALERT MANAGEMENT
-- ============================================================

CREATE TABLE alert_rules (
    alert_rule_id       UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,

    growth_stage_id     UUID REFERENCES growth_stages(growth_stage_id) ON DELETE SET NULL,

    measurement_type_id UUID NOT NULL REFERENCES measurement_types(measurement_type_id) ON DELETE RESTRICT,

    rule_name           VARCHAR(150) NOT NULL,

    min_threshold       NUMERIC(14,4),
    max_threshold       NUMERIC(14,4),

    severity             VARCHAR(30) NOT NULL DEFAULT 'WARNING',

    is_active           BOOLEAN NOT NULL DEFAULT TRUE,

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE alerts (
    alert_id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,
    sensor_id           UUID REFERENCES sensors(sensor_id) ON DELETE SET NULL,
    alert_rule_id       UUID REFERENCES alert_rules(alert_rule_id) ON DELETE SET NULL,

    title               VARCHAR(200) NOT NULL,
    message             TEXT NOT NULL,

    severity            VARCHAR(30) NOT NULL,

    triggered_value     NUMERIC(14,4),

    status              VARCHAR(30) NOT NULL DEFAULT 'OPEN',

    triggered_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    acknowledged_by     UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    acknowledged_at     TIMESTAMPTZ,

    resolved_by         UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    resolved_at         TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE notifications (
    notification_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id             UUID NOT NULL REFERENCES app_users(user_id) ON DELETE CASCADE,

    alert_id            UUID REFERENCES alerts(alert_id) ON DELETE CASCADE,

    title               VARCHAR(200) NOT NULL,
    message             TEXT NOT NULL,

    channel             VARCHAR(30) NOT NULL DEFAULT 'IN_APP',

    is_read             BOOLEAN NOT NULL DEFAULT FALSE,

    sent_at             TIMESTAMPTZ,
    read_at             TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 11. CONTROL / AUTOMATION
-- ============================================================

CREATE TABLE control_rules (
    control_rule_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,

    name                VARCHAR(150) NOT NULL,
    description         TEXT,

    measurement_type_id UUID REFERENCES measurement_types(measurement_type_id) ON DELETE SET NULL,

    condition_operator  VARCHAR(20),

    threshold_value     NUMERIC(14,4),

    actuator_id         UUID REFERENCES actuators(actuator_id) ON DELETE SET NULL,

    action_type         VARCHAR(50),

    action_value        NUMERIC(14,4),

    is_active           BOOLEAN NOT NULL DEFAULT TRUE,

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE control_schedules (
    schedule_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID NOT NULL REFERENCES zones(zone_id) ON DELETE CASCADE,
    actuator_id         UUID NOT NULL REFERENCES actuators(actuator_id) ON DELETE CASCADE,

    name                VARCHAR(150) NOT NULL,

    start_time          TIME NOT NULL,
    end_time            TIME,

    days_of_week        VARCHAR(50),

    action_type         VARCHAR(50) NOT NULL,

    action_value        NUMERIC(14,4),

    is_active           BOOLEAN NOT NULL DEFAULT TRUE,

    created_by          UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE control_commands (
    command_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,
    actuator_id         UUID NOT NULL REFERENCES actuators(actuator_id) ON DELETE CASCADE,

    requested_by        UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    source_type         VARCHAR(30) NOT NULL,
    -- MANUAL
    -- SCHEDULE
    -- RULE
    -- AI_RECOMMENDATION

    action_type         VARCHAR(50) NOT NULL,

    action_value        NUMERIC(14,4),

    requested_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    status              VARCHAR(30) NOT NULL DEFAULT 'PENDING',

    executed_at         TIMESTAMPTZ
);


CREATE TABLE control_feedback (
    feedback_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    command_id          UUID NOT NULL REFERENCES control_commands(command_id) ON DELETE CASCADE,

    feedback_status     VARCHAR(30) NOT NULL,

    actual_value        NUMERIC(14,4),

    message             TEXT,

    received_at         TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 12. WEATHER
-- ============================================================

CREATE TABLE weather_records (
    weather_record_id   BIGSERIAL PRIMARY KEY,

    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    recorded_at         TIMESTAMPTZ NOT NULL,

    temperature         NUMERIC(8,2),
    humidity            NUMERIC(8,2),

    rainfall_mm         NUMERIC(10,2),

    wind_speed          NUMERIC(10,2),
    weather_condition   VARCHAR(100),

    source              VARCHAR(100),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE weather_forecasts (
    forecast_id         BIGSERIAL PRIMARY KEY,

    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    forecast_time       TIMESTAMPTZ NOT NULL,

    temperature         NUMERIC(8,2),
    humidity            NUMERIC(8,2),

    rainfall_probability NUMERIC(5,2),
    rainfall_mm         NUMERIC(10,2),

    weather_condition   VARCHAR(100),

    source              VARCHAR(100),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 13. AI ADVISORY
-- ============================================================

CREATE TABLE ai_conversations (
    conversation_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id             UUID NOT NULL REFERENCES app_users(user_id) ON DELETE CASCADE,

    farm_id             UUID REFERENCES farms(farm_id) ON DELETE SET NULL,
    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,

    title               VARCHAR(200),

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE ai_messages (
    message_id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    conversation_id     UUID NOT NULL REFERENCES ai_conversations(conversation_id) ON DELETE CASCADE,

    sender_type         VARCHAR(30) NOT NULL,
    -- USER
    -- AI

    message_text       TEXT NOT NULL,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE ai_recommendations (
    recommendation_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    user_id             UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    farm_id             UUID REFERENCES farms(farm_id) ON DELETE SET NULL,
    zone_id             UUID REFERENCES zones(zone_id) ON DELETE SET NULL,

    planting_season_id  UUID REFERENCES planting_seasons(planting_season_id)
                        ON DELETE SET NULL,

    conversation_id     UUID REFERENCES ai_conversations(conversation_id)
                        ON DELETE SET NULL,

    recommendation_type VARCHAR(50) NOT NULL,

    title               VARCHAR(200) NOT NULL,

    recommendation_text TEXT NOT NULL,

    reasoning           TEXT,

    status              VARCHAR(30) NOT NULL DEFAULT 'PENDING',

    accepted_at         TIMESTAMPTZ,
    rejected_at         TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 14. MATERIAL / INVENTORY
-- ============================================================

CREATE TABLE material_categories (
    material_category_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    name                VARCHAR(100) NOT NULL UNIQUE,
    description         TEXT
);

-- Example:
-- SEED
-- FERTILIZER
-- PESTICIDE
-- GROWING_MEDIA
-- OTHER


CREATE TABLE materials (
    material_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    tenant_id           UUID NOT NULL REFERENCES tenants(tenant_id) ON DELETE CASCADE,

    material_category_id UUID
        REFERENCES material_categories(material_category_id)
        ON DELETE SET NULL,

    name                VARCHAR(150) NOT NULL,
    code                VARCHAR(50) NOT NULL,

    unit                VARCHAR(30) NOT NULL,

    description         TEXT,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE (tenant_id, code)
);


CREATE TABLE crop_material_requirements (

    unit                VARCHAR(30),

CREATE TABLE inventories (

    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    UNIQUE (farm_id, material_id)

CREATE TABLE material_usages (

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()

-- ============================================================
-- 17. DEVICE MAINTENANCE / REPLACEMENT
-- ============================================================

CREATE TABLE service_requests (
    service_request_id  UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    tenant_id           UUID NOT NULL REFERENCES tenants(tenant_id) ON DELETE CASCADE,
    farm_id             UUID NOT NULL REFERENCES farms(farm_id) ON DELETE CASCADE,

    gateway_id          UUID REFERENCES gateways(gateway_id) ON DELETE SET NULL,
    node_id             UUID REFERENCES sensor_nodes(node_id) ON DELETE SET NULL,
    sensor_id           UUID REFERENCES sensors(sensor_id) ON DELETE SET NULL,
    actuator_id         UUID REFERENCES actuators(actuator_id) ON DELETE SET NULL,

    requested_by        UUID REFERENCES app_users(user_id) ON DELETE SET NULL,
    assigned_technician UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    title               VARCHAR(200) NOT NULL,
    description         TEXT NOT NULL,

    priority            VARCHAR(30) NOT NULL DEFAULT 'MEDIUM',

    status              VARCHAR(30) NOT NULL DEFAULT 'OPEN',

    request_type        VARCHAR(30) NOT NULL DEFAULT 'MAINTENANCE',

    -- MAINTENANCE
    -- REPAIR
    -- REPLACEMENT

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    assigned_at         TIMESTAMPTZ,
    completed_at        TIMESTAMPTZ,

    resolution          TEXT
);


CREATE TABLE maintenance_records (
    maintenance_record_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    service_request_id  UUID NOT NULL
        REFERENCES service_requests(service_request_id)
        ON DELETE CASCADE,

    technician_id       UUID REFERENCES app_users(user_id) ON DELETE SET NULL,

    maintenance_type   VARCHAR(50) NOT NULL,

    diagnosis           TEXT,
    action_taken        TEXT,

    cost                NUMERIC(14,2),

    started_at          TIMESTAMPTZ,
    completed_at        TIMESTAMPTZ,

    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE TABLE device_replacements (
    replacement_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    service_request_id  UUID NOT NULL
        REFERENCES service_requests(service_request_id)
        ON DELETE CASCADE,

    old_node_id         UUID REFERENCES sensor_nodes(node_id) ON DELETE SET NULL,
    old_sensor_id       UUID REFERENCES sensors(sensor_id) ON DELETE SET NULL,
    old_actuator_id     UUID REFERENCES actuators(actuator_id) ON DELETE SET NULL,
    old_gateway_id      UUID REFERENCES gateways(gateway_id) ON DELETE SET NULL,

    reason              TEXT,

    replacement_date    TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    technician_id       UUID REFERENCES app_users(user_id) ON DELETE SET NULL
);
